#!/usr/bin/env bash
# Reusable NKP management-cluster environment template.
# Method: air-gapped local bundle deployment on Nutanix AHV.
# Harbor / external registry mirror: not used.
# Replace every CHANGE_ME value before running the deployment script.

# -----------------------------------------------------------------------------
# NKP release and extracted air-gapped bundle paths
# Expected layout:
#   /data/nkp-<version>/bundles/airgapped/nkp-v<version>/
# -----------------------------------------------------------------------------
export NKP_VERSION="2.16.1"
export NKP_BASE_DIR="/data/nkp-${NKP_VERSION}"
export NKP_BUNDLE_DIR="${NKP_BASE_DIR}/bundles/airgapped/nkp-v${NKP_VERSION}"

export NKP_BOOTSTRAP_IMAGE="${NKP_BUNDLE_DIR}/konvoy-bootstrap-image-v${NKP_VERSION}.tar"
export NKP_KONVOY_BUNDLE="${NKP_BUNDLE_DIR}/container-images/konvoy-image-bundle-v${NKP_VERSION}.tar"
export NKP_KOMMANDER_BUNDLE="${NKP_BUNDLE_DIR}/container-images/kommander-image-bundle-v${NKP_VERSION}.tar"

# -----------------------------------------------------------------------------
# NKP management cluster
# CLUSTER_NAME must be lowercase and RFC 1123 compliant.
# Set CLUSTER_HOSTNAME to the NKP ingress/LoadBalancer IP when not supplying
# custom ingress certificates.
# -----------------------------------------------------------------------------
export CLUSTER_NAME="CHANGE_ME_LOWERCASE_CLUSTER_NAME"
export CLUSTER_HOSTNAME="CHANGE_ME_LB_IP"

# -----------------------------------------------------------------------------
# Prism Central and AHV resources
# -----------------------------------------------------------------------------
export NUTANIX_PC_FQDN_ENDPOINT_WITH_PORT="https://CHANGE_ME_PC_IP_OR_FQDN:9440"
export NUTANIX_USER="CHANGE_ME_PC_USERNAME"

# Supply this securely from the shell instead of saving it in the file.
export NUTANIX_PASSWORD="${NUTANIX_PASSWORD:-}"

export PRISM_ELEMENT_CLUSTER_NAME="CHANGE_ME_PE_CLUSTER_NAME"
export SUBNET_NAME="CHANGE_ME_AHV_SUBNET_NAME"
export STORAGE_CONTAINER="CHANGE_ME_STORAGE_CONTAINER"
export IMAGE_NAME="CHANGE_ME_NKP_NODE_IMAGE.qcow2"

# -----------------------------------------------------------------------------
# Node counts
# -----------------------------------------------------------------------------

export WORKER_REPLICAS="4"

# -----------------------------------------------------------------------------
# Kubernetes networking
# These CIDRs must not overlap with the physical network, VPN, Docker networks,
# or other Kubernetes networks reachable from this environment.

2. Reusable NKP management-cluster command

The essential extracted deployment command is:

nkp create cluster nutanix \
  --cluster-name "${CLUSTER_NAME}" \
  --cluster-hostname "${CLUSTER_HOSTNAME}" \
  --self-managed \
  --airgapped \
  --bootstrap-cluster-image "${NKP_BOOTSTRAP_IMAGE}" \
  --bundle "${NKP_KONVOY_BUNDLE}" \
  --bundle "${NKP_KOMMANDER_BUNDLE}" \
  --endpoint "${NUTANIX_PC_FQDN_ENDPOINT_WITH_PORT}" \
  --insecure \
  --control-plane-endpoint-ip "${CONTROL_PLANE_IP}" \
  --control-plane-vm-image "${IMAGE_NAME}" \
  --control-plane-prism-element-cluster "${PRISM_ELEMENT_CLUSTER_NAME}" \
  --control-plane-subnets "${SUBNET_NAME}" \
  --control-plane-replicas "${CONTROL_PLANE_REPLICAS}" \
  --worker-vm-image "${IMAGE_NAME}" \
  --worker-prism-element-cluster "${PRISM_ELEMENT_CLUSTER_NAME}" \
  --worker-subnets "${SUBNET_NAME}" \
  --worker-replicas "${WORKER_REPLICAS}" \
  --csi-storage-container "${STORAGE_CONTAINER}" \
  --kubernetes-pod-network-cidr "${POD_CIDR}" \
  --kubernetes-service-cidr "${SERVICE_CIDR}" \
  --kubernetes-service-load-balancer-ip-range "${LB_IP_RANGE}" \
  --ssh-username "${SSH_USERNAME}" \
  --ssh-public-key-file "${SSH_KEY_FILE}" \
  --verbose 5

The downloaded deployment script wraps this command with checks for missing variables, missing bundle files, an invalid uppercase cluster name, a missing SSH key, and a missing NKP CLI.

What makes this the no-Harbor bundle approach

The deployment supplies the required images directly from the Bastion through:

--bootstrap-cluster-image "${NKP_BOOTSTRAP_IMAGE}"
--bundle "${NKP_KONVOY_BUNDLE}"
--bundle "${NKP_KOMMANDER_BUNDLE}"
--airgapped

It deliberately contains none of the following:

--registry-mirror-url
--registry-mirror-username
--registry-mirror-password
--registry-mirror-cacert
Harbor hostname
Harbor project
Harbor credentials

It also excludes the custom certificate options that were removed in the later successful deployment:

--ingress-certificate
--ingress-private-key
--ingress-ca

This package is specifically for creating the self-managed NKP management cluster. The --self-managed flag and Kommander bundle are retained from the working management-cluster flow. cd /data/nkp-2.16.1/scripts

# Load the reusable project variables
source ./env_nkp_mgmt.sh

# Prompt for the Prism Central password when it is not already exported
if [[ -z "${NUTANIX_PASSWORD:-}" ]]; then
  read -rsp "Enter Prism Central password: " NUTANIX_PASSWORD

# Secure directory for generated manifests

# Native NKP dry run: generate manifests only
nkp create cluster nutanix \

  --cluster-hostname "${CLUSTER_HOSTNAME}" \
  --self-managed \
  --airgapped \

  --bundle "${NKP_KONVOY_BUNDLE}" \
  --bundle "${NKP_KOMMANDER_BUNDLE}" \
  --endpoint "${NUTANIX_PC_FQDN_ENDPOINT_WITH_PORT}" \
  --insecure \
  --control-plane-endpoint-ip "${CONTROL_PLANE_IP}" \
  --control-plane-vm-image "${IMAGE_NAME}" \
  --control-plane-prism-element-cluster "${PRISM_ELEMENT_CLUSTER_NAME}" \
  --control-plane-subnets "${SUBNET_NAME}" \
  --control-plane-replicas "${CONTROL_PLANE_REPLICAS}" \
  --worker-vm-image "${IMAGE_NAME}" \
  --worker-prism-element-cluster "${PRISM_ELEMENT_CLUSTER_NAME}" \
  --worker-subnets "${SUBNET_NAME}" \
  --worker-replicas "${WORKER_REPLICAS}" \
  --csi-storage-container "${STORAGE_CONTAINER}" \
  --kubernetes-pod-network-cidr "${POD_CIDR}" \
  --kubernetes-service-cidr "${SERVICE_CIDR}" \
  --kubernetes-service-load-balancer-ip-range "${LB_IP_RANGE}" \
  --ssh-username "${SSH_USERNAME}" \
  --ssh-public-key-file "${SSH_KEY_FILE}" \
  --dry-run \
  -o yaml \
  --output-directory "${DRY_RUN_DIR}" \

Important security note

The dry run can generate Prism Central credential Secret manifests along with the Cluster API manifests. Keep the directory restricted and do not email, upload, or commit it to Git. The Nutanix example shows outputs such as the cluster YAML and Prism credential Secret YAML files. (Nutanix)

Confirm permissions:

chmod -R go-rwx "${DRY_RUN_DIR}"
find "${DRY_RUN_DIR}" -maxdepth 1 -type f -exec ls -l {} \;

Do not run this command unless you intentionally want to begin provisioning:

kubectl apply -f "${DRY_RUN_DIR}"

The native dry run will generate the deployment manifests, but it will not create the target NKP management-cluster VMs on AHV.
